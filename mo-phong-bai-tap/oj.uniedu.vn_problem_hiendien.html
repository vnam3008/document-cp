🔎 Tìm Ma Trận Con (Tọa độ 1-based)

Tốc độ: Vừa

M x N Ma Trận A (Tranh)

P x Q Khung B (Mẫu)

💻 Code C++

Đồng bộ mô phỏng 
for (int i = 1; i <= M - P + 1; i++) {


  for (int j = 1; j <= N - Q + 1; j++) {


    // Đặt khung B lên A tại tọa độ (i, j)


    bool isMatch = true;


    for (int x = 1; x <= P; x++) {


      for (int y = 1; y <= Q; y++) {


        int rA = i + x - 1; // Hàng tương ứng trên A


        int cA = j + y - 1; // Cột tương ứng trên A


        if (A[rA][cA] != B[x][y]) {


          isMatch = false; // SAI!


          break; // Hủy khung này


        }


      }


      if (!isMatch) break;


    }


    if (isMatch) return true; // TÌM THẤY!


  }


}


return false; // Không thấy
